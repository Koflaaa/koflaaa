using System;

namespace RateSpiel
{
    class Program
    {
        static void Main(string[] args)
        {
            Random random = new Random();
            int c = random.Next(1, 101);

            int y = 0;
            int x = 0;

            while (x != c)
            {
                Console.Write("Bitte gib eine Zahl zwischen 1 und 100 ein: ");
                x = Convert.ToInt32(Console.ReadLine());
                y++;

                if (x < c)
                {
                    Console.WriteLine("Die eingegebene Zahl ist zu klein. Versuche es erneut.");
                }
                else if (x > c)
                {
                    Console.WriteLine("Die eingegebene Zahl ist zu groß. Versuche es erneut.");
                }
            }

            Console.WriteLine("Die gesuchte Zahl war {0}.", c);
            Console.WriteLine("Du hast insgesamt {0} Versuche gebraucht.", y);
            Console.WriteLine("Drücke eine beliebige Taste, um das Spiel zu beenden.");
            Console.ReadKey();
        }
    }
}
